#include <iostream>
using namespace std;

struct Node{
    int data;
    Node *left;
    Node *right;
    
    Node(int val){
        data = val;
        left = right = nullptr;
    }
};

Node* removeNode(Node* targetNode, int value){
    if(targetNode == nullptr){
        cout << "Error not found." << endl;
    } 
    
    if(value < targetNode->data){
        targetNode -> left = removeNode(targetNode->left, value);
    } else if(value > targetNode->data){
        targetNode -> right = removeNode(targetNode->right, value);
    } else {
        if(targetNode -> left == nullptr && targetNode->right == nullptr){
            delete targetNode;
            return nullptr;
        } else if(targetNode -> left == nullptr){
            Node *temp = targetNode -> right;
            delete targetNode;
            return temp;
        } else if(targetNode -> right == nullptr){
            Node *temp = targetNode -> left;
            delete targetNode;
            return temp;
        } else {
            Node *temp = targetNode->left;
            while(temp->right != nullptr){
                temp = temp-> right;
            }
            targetNode->data = temp->data;
            targetNode->left = removeNode(targetNode->left, temp->data);
        }
        
    }
    return targetNode;
    
}

Node* insert(Node*root, int val){
    if(root == nullptr){
        return new Node(val);
    }
    
    if(val < root->data){
        root->left = insert(root->left, val);
    } else if(val > root->data){
        root->right = insert(root->right, val);
    }
    
    return root;
}

void print(Node *root){
    if(root != nullptr){
        cout << root->data << " " << endl;
        print(root->left);
        print(root->right);
    }
}
